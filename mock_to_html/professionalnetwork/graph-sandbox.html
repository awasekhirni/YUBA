// Find users with strongest mutual connections to you
MATCH (me:User {id: 42})-[c1:CONNECTED]-(peer:User)-[c2:CONNECTED]-(me)
WHERE peer.trust_score >= 85
RETURN peer.name,
  count(c1) + count(c2) AS mutual_strength,
  peer.industry, peer.location
ORDER BY mutual_strength DESC
LIMIT 25;
