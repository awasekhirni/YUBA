Quick-Start Code · JS SDK📋 Copy
// npm install @yuba/sdk
import { YubaClient } from '@yuba/sdk';

const yuba = new YubaClient({
  apiKey: process.env.YUBA_API_KEY,
  environment: 'staging',
  idempotencyStore: 'redis'
});

// Fetch network + trust metrics
const connections = await yuba.networks.get(42, {
  expand: ['trust_score', 'endorsements'],
  include: 'mutuals'
});

// Trigger recommendation engine
const recs = await yuba.ai.recommend({
  model: 'career-path-v3',
  userId: 42,
  explain: true
});
