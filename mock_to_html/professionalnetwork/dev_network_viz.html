// Find mutual connections with high trust
MATCH (me:User {id: 42})-[:CONNECTED*1..2]-(peer:User)
WHERE peer.trust_score > 85
  AND peer.industry = 'Legal'
RETURN peer.name, peer.trust_score,
       size((me)-[:CONNECTED]-(peer)) AS mutual
ORDER BY peer.trust_score DESC
LIMIT 50;
